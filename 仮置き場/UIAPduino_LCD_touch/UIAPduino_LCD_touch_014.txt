#include <Arduino.h>
const uint32_t CH_OUT   = 0x1; // 出力 push-pull
const uint32_t CH_IN_PU = 0x8; // 入力 プルアップ/プルダウン(方向はODRビットで決定)
const int TOUCH_MIN = 90; // ★ADC方式化に伴いスケールが変わるため要再調整
const int SENSE_BIT = 2;  // PA2
const int ADC_CHANNEL = 0; // PA2 = ADC1チャンネル0
const int ADC_SAMPLE_TIME_CODE = 1; // 最速に近いサンプル時間(充電初期の電圧を捉えるため)
const unsigned long STABLE_MS = 400;
const unsigned long WIN_START_MS = 200;
const int SMOOTH_N = 16;
int baseA0 = 0;
int leftVal = 0;
int rightVal = 0;
int calState = 0;
unsigned long phaseStart = 0;
long calSum = 0;
int calCount = 0;
int smoothBuf[SMOOTH_N];
int smoothIdx = 0;
int smoothCount = 0;
long smoothSum = 0;
void setupHardwareUart() {
  RCC->APB2PCENR |= (RCC_APB2Periph_GPIOD | RCC_APB2Periph_AFIO | RCC_APB2Periph_USART1);
  GPIOD->CFGLR &= ~((0xFu << (5 * 4)) | (0xFu << (6 * 4)));
  GPIOD->CFGLR |=  ((0xBu << (5 * 4)) | (0x8u << (6 * 4)));
  GPIOD->OUTDR |= (1 << 6);
  AFIO->PCFR1 &= ~(1 << 2);
  USART1->CTLR1 &= ~USART_CTLR1_UE;
  USART1->BRR = 0x1A1;
  USART1->CTLR1 = USART_CTLR1_UE | USART_CTLR1_TE | USART_CTLR1_RE;
}
void uartSendChar(char c) {
  while (!(USART1->STATR & (1 << 7)));
  USART1->DATAR = c;
}
void uartSendString(const char* str) {
  while (*str) uartSendChar(*str++);
}
void uartSendInt(int v) {
  if (v < 0) {
    uartSendChar('-');
    v = -v;
  }
  char buf[12];
  uint8_t i = 0;
  if (v == 0) {
    uartSendChar('0');
    return;
  }
  while (v > 0) {
    buf[i++] = '0' + (v % 10);
    v /= 10;
  }
  while (i > 0) uartSendChar(buf[--i]);
}
void setupPwmPc3() {
  RCC->APB2PCENR |= RCC_APB2Periph_TIM1;
  GPIOC->CFGLR &= ~(0xFu << (3 * 4));
  GPIOC->CFGLR |=  (0xBu << (3 * 4));
  TIM1->CTLR1 = 0;
  TIM1->PSC = 47;
  TIM1->ATRLR = 999;
  TIM1->CHCTLR2 = 0;
  TIM1->CHCTLR2 = (6u << 4) | (1u << 3);
  TIM1->CCER |= (1u << 8);
  TIM1->CH3CVR = 0;
  TIM1->BDTR |= (1u << 15);
  TIM1->CTLR1 |= 1u;
}
void setPwmPercent(int percent) {
  if (percent < 0) percent = 0;
  if (percent > 100) percent = 100;
  TIM1->CH3CVR = (uint16_t)(((uint32_t)(TIM1->ATRLR + 1) * (uint32_t)percent) / 100u);
}
void blinkLed(int times) {
  for (int i = 0; i < times; i++) {
    GPIOC->BSHR = 1;
    delay(120);
    GPIOC->BCR = 1;
    delay(120);
  }
}
void setupAdcTouch() {
  RCC->APB2PCENR |= RCC_APB2Periph_ADC1;
  ADC1->RSQR3 = ADC_CHANNEL;
  ADC1->SAMPTR2 = ((uint32_t)ADC_SAMPLE_TIME_CODE) << (3 * ADC_CHANNEL);
  ADC1->CTLR2 = ADC_ADON;
  delay(2); // ADC安定待ち
}
// PA2を放電→内蔵プルアップの入力へ切替→充電初期の電圧をADCで1回サンプリング
int adcSampleTouch(int bit) {
  uint32_t sh = bit * 4;
  // 1) 出力LOWで放電
  GPIOA->CFGLR &= ~(0xFu << sh);
  GPIOA->CFGLR |=  (CH_OUT << sh);
  GPIOA->BCR = (1 << bit);
  delayMicroseconds(80);
  // 2) 入力(内蔵プルアップ)へ切替。ODR=1でプルアップ方向
  GPIOA->CFGLR &= ~(0xFu << sh);
  GPIOA->CFGLR |=  (CH_IN_PU << sh);
  GPIOA->OUTDR |= (1 << bit);
  // 3) 充電が始まった直後にADCサンプリング
  ADC1->CTLR2 = ADC_SWSTART | ADC_ADON | ADC_EXTSEL;
  while (!(ADC1->STATR & ADC_EOC));
  int v = ADC1->RDATAR;
  return v;
}
int readCap(int bit) {
  long s = 0;
  for (int i = 0; i < 32; i++) s += adcSampleTouch(bit);
  return (int)s;
}
int isTouch(int raw) {
  int d = raw - baseA0;
  if (d < 0) d = 0;
  return (d >= TOUCH_MIN) ? 1 : 0;
}
int rawToPercent(int raw) {
  if (leftVal == rightVal) return 50;
  long p = ((long)(raw - leftVal) * 100L) / (long)(rightVal - leftVal);
  if (p < 0) p = 0;
  if (p > 100) p = 100;
  return (int)p;
}
void smoothReset() {
  smoothIdx = 0;
  smoothCount = 0;
  smoothSum = 0;
}
int smoothPush(int raw) {
  if (smoothCount < SMOOTH_N) {
    smoothBuf[smoothIdx] = raw;
    smoothSum += raw;
    smoothCount++;
  } else {
    smoothSum -= smoothBuf[smoothIdx];
    smoothSum += raw;
    smoothBuf[smoothIdx] = raw;
  }
  smoothIdx = (smoothIdx + 1) % SMOOTH_N;
  return (int)(smoothSum / smoothCount);
}
void setup() {
  setupHardwareUart();
  RCC->APB2PCENR |= (RCC_APB2Periph_GPIOA | RCC_APB2Periph_GPIOC);
  GPIOC->CFGLR &= ~0xFu;
  GPIOC->CFGLR |= CH_OUT;
  setupPwmPc3();
  setPwmPercent(0);
  setupAdcTouch();
  uartSendString("=== LCD slider Ver2.04 (ADC touch) ===\r\n");
  uartSendString("Calibrating OFF...\r\n");
  delay(400);
  for (int i = 0; i < 5; i++) {
    readCap(SENSE_BIT);
    delay(10);
  }
  baseA0 = readCap(SENSE_BIT);
  uartSendString("Base A0=");
  uartSendInt(baseA0);
  uartSendString("\r\n");
  blinkLed(1);
  calState = 1;
  uartSendString("Touch LEFT\r\n");
}
void loop() {
  int raw = readCap(SENSE_BIT);
  unsigned long now = millis();
  if (calState == 1) {
    if (isTouch(raw)) {
      phaseStart = now;
      calSum = 0;
      calCount = 0;
      calState = 2;
    }
    delay(10);
    return;
  }
  if (calState == 2) {
    if (!isTouch(raw)) {
      // 400ms未満で離れた=早期離脱。左タッチ待ちへ戻る
      calState = 1;
      delay(10);
      return;
    }
    unsigned long elapsed = now - phaseStart;
    if (elapsed >= WIN_START_MS && elapsed <= STABLE_MS) {
      calSum += raw;
      calCount++;
    }
    if (elapsed < STABLE_MS) {
      delay(10);
      return;
    }
    // 400ms経過。まだ触れている状態でここで確定する
    leftVal = (calCount > 0) ? (int)(calSum / calCount) : raw;
    uartSendString("LEFT=");
    uartSendInt(leftVal);
    uartSendString("\r\n");
    blinkLed(2);
    phaseStart = millis();
    calState = 3;
    uartSendString("Release wait\r\n");
    delay(10);
    return;
  }
  if (calState == 3) {
    if (isTouch(raw)) {
      phaseStart = now;
    } else if ((now - phaseStart) >= STABLE_MS) {
      calState = 4;
      uartSendString("Touch RIGHT\r\n");
    }
    delay(10);
    return;
  }
  if (calState == 4) {
    if (isTouch(raw)) {
      phaseStart = now;
      calSum = 0;
      calCount = 0;
      calState = 5;
    }
    delay(10);
    return;
  }
  if (calState == 5) {
    if (!isTouch(raw)) {
      // 400ms未満で離れた=早期離脱。右タッチ待ちへ戻る
      calState = 4;
      delay(10);
      return;
    }
    unsigned long elapsed = now - phaseStart;
    if (elapsed >= WIN_START_MS && elapsed <= STABLE_MS) {
      calSum += raw;
      calCount++;
    }
    if (elapsed < STABLE_MS) {
      delay(10);
      return;
    }
    // 400ms経過。まだ触れている状態でここで確定する
    rightVal = (calCount > 0) ? (int)(calSum / calCount) : raw;
    uartSendString("RIGHT=");
    uartSendInt(rightVal);
    uartSendString("\r\n");
    blinkLed(3);
    phaseStart = millis();
    calState = 6;
    uartSendString("Release wait\r\n");
    delay(10);
    return;
  }
  if (calState == 6) {
    if (isTouch(raw)) {
      phaseStart = now;
    } else if ((now - phaseStart) >= STABLE_MS) {
      calState = 7;
      smoothReset();
      uartSendString("RUN\r\n");
    }
    delay(10);
    return;
  }
  int currentState = 0;
  const char* pos = "NONE ";
  int sliderPos = 0;
  if (!isTouch(raw)) {
    currentState = 0;
    pos = "NONE ";
    smoothReset();
    GPIOC->BCR = 1;
    setPwmPercent(0);
  } else {
    currentState = 1;
    pos = "TOUCH";
    int avgRaw = smoothPush(raw);
    sliderPos = rawToPercent(avgRaw);
    GPIOC->BSHR = 1;
    setPwmPercent(sliderPos);
  }
  static int printTimer = 0;
  printTimer++;
  if (printTimer >= 50) {
    uartSendString("raw=");
    uartSendInt(raw);
    uartSendString(" d=");
    uartSendInt(raw - baseA0);
    uartSendChar(' ');
    uartSendString(pos);
    if (currentState != 0) {
      uartSendString(" POS=");
      uartSendInt(sliderPos);
      uartSendString("%");
    }
    uartSendString("\r\n");
    printTimer = 0;
  }
  delay(10);
}
