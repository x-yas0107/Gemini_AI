#include <Arduino.h>

const uint32_t CH_OUT   = 0x1; // 出力 push-pull
const uint32_t CH_IN_FL = 0x4; // 入力 フローティング(LCD駆動非アクティブ時用)
const uint32_t CH_IN_PU = 0x8; // 入力 プルアップ/プルダウン(方向はODRビットで決定)

const int TOUCH_MIN = 150; 
const int SENSE_BIT = 2;   // PA2
const int ADC_CHANNEL = 0; // PA2 = ADC1チャンネル0
const int ADC_SAMPLE_TIME_CODE = 1; 

const unsigned long STABLE_MS = 400;
const unsigned long WIN_START_MS = 200;
const int SMOOTH_N = 16;

int baseA0 = 0;
int maxVal = 0;
int calState = 0;
unsigned long phaseStart = 0;
long calSum = 0;
int calCount = 0;

int smoothBuf[SMOOTH_N];
int smoothIdx = 0;
int smoothCount = 0;
long smoothSum = 0;

// LCD表示用変数
int dispNum = 0;

/*
 * 実配線に基づくフォントマッピング(0〜F)
 * bit 0: PC1+PC4 = f
 * bit 1: PC1+PC5 = a
 * bit 2: PC1+PC6 = b
 * bit 3: PA2+PC4 = e
 * bit 4: PA2+PC5 = g
 * bit 5: PA2+PC6 = c
 * bit 6: PC2+PC4 = 未使用 (常に0)
 * bit 7: PC2+PC5 = d
 * bit 8: PC2+PC6 = h (ドット)
 */
const uint16_t font3x3[16] = {
  0b010101111, // 0: a,b,c,d,e,f
  0b000100100, // 1: b,c
  0b010011110, // 2: a,b,d,e,g
  0b010110110, // 3: a,b,c,d,g
  0b000110101, // 4: b,c,f,g
  0b010110011, // 5: a,c,d,f,g
  0b010111011, // 6: a,c,d,e,f,g (6)
  0b000100110, // 7: a,b,c
  0b010111111, // 8: a,b,c,d,e,f,g
  0b010110111, // 9: a,b,c,d,f,g
  0b000111111, // A: a,b,c,e,f,g
  0b010111001, // b: c,d,e,f,g
  0b010001011, // C: a,d,e,f
  0b010111100, // d: b,c,d,e,g
  0b010011011, // E: a,d,e,f,g
  0b000011011  // F: a,e,f,g
};

void setupHardwareUart() {
  RCC->APB2PCENR |= (RCC_APB2Periph_GPIOD | RCC_APB2Periph_AFIO | RCC_APB2Periph_USART1);
  GPIOD->CFGLR &= ~((0xFu << (5 * 4)) | (0xFu << (6 * 4)));
  GPIOD->CFGLR |=  ((0xBu << (5 * 4)) | (0x8u << (6 * 4)));
  GPIOD->OUTDR |= (1 << 6);
  AFIO->PCFR1 &= ~(1 << 2);
  USART1->CTLR1 &= ~USART_CTLR1_UE;
  USART1->BRR = 0x1A1;
  USART1->CTLR1 = USART_CTLR1_UE | USART_CTLR1_TE | USART_CTLR1_RE;
}

void uartSendChar(char c) {
  while (!(USART1->STATR & (1 << 7)));
  USART1->DATAR = c;
}

void uartSendString(const char* str) {
  while (*str) uartSendChar(*str++);
}

void uartSendInt(int v) {
  if (v < 0) {
    uartSendChar('-');
    v = -v;
  }
  char buf[12];
  uint8_t i = 0;
  if (v == 0) {
    uartSendChar('0');
    return;
  }
  while (v > 0) {
    buf[i++] = '0' + (v % 10);
    v /= 10;
  }
  while (i > 0) uartSendChar(buf[--i]);
}

void setupPwmPc3() {
  RCC->APB2PCENR |= RCC_APB2Periph_TIM1;
  GPIOC->CFGLR &= ~(0xFu << (3 * 4));
  GPIOC->CFGLR |=  (0xBu << (3 * 4));
  TIM1->CTLR1 = 0;
  TIM1->PSC = 47;
  TIM1->ATRLR = 999;
  TIM1->CHCTLR2 = 0;
  TIM1->CHCTLR2 = (6u << 4) | (1u << 3);
  TIM1->CCER |= (1u << 8);
  TIM1->CH3CVR = 0;
  TIM1->BDTR |= (1u << 15);
  TIM1->CTLR1 |= 1u;
}

void setPwmPercent(int percent) {
  if (percent < 0) percent = 0;
  if (percent > 100) percent = 100;
  TIM1->CH3CVR = (uint16_t)(((uint32_t)(TIM1->ATRLR + 1) * (uint32_t)percent) / 100u);
}

void setupAdcTouch() {
  RCC->APB2PCENR |= RCC_APB2Periph_ADC1;
  ADC1->RSQR3 = ADC_CHANNEL;
  ADC1->SAMPTR2 = ((uint32_t)ADC_SAMPLE_TIME_CODE) << (3 * ADC_CHANNEL);
  ADC1->CTLR2 = ADC_ADON;
  delay(2);
}

void UpdateLCD(int num) {
  uint16_t pat = font3x3[num & 0x0F];
  
  // 全COM・SEGピンを一旦Push-Pull出力に設定
  GPIOA->CFGLR = (GPIOA->CFGLR & ~(0xFu << (2 * 4))) | (CH_OUT << (2 * 4));
  uint32_t c_clr = GPIOC->CFGLR & ~((0xFu << (1 * 4)) | (0xFu << (2 * 4)) | (0xFu << (4 * 4)) | (0xFu << (5 * 4)) | (0xFu << (6 * 4)));
  GPIOC->CFGLR = c_clr | (CH_OUT << (1 * 4)) | (CH_OUT << (2 * 4)) | (CH_OUT << (4 * 4)) | (CH_OUT << (5 * 4)) | (CH_OUT << (6 * 4));
  
  for(int phase = 0; phase < 2; phase++) { 
    for(int c = 0; c < 3; c++) {           
      int segBits = (pat >> (c * 3)) & 0x07;
      int comState = phase; 
      
      uint32_t bshrC = 0, bcrC = 0;
      
      // アクティブなCOMの出力レベル設定
      if (c == 0) { if (comState) bshrC |= (1 << 1); else bcrC |= (1 << 1); }
      if (c == 1) { if (comState) GPIOA->BSHR = (1 << 2); else GPIOA->BCR = (1 << 2); }
      if (c == 2) { if (comState) bshrC |= (1 << 2); else bcrC |= (1 << 2); }
      
      // SEGの出力レベル設定
      for(int s = 0; s < 3; s++) {
        int isActive = (segBits & (1 << s)) ? 1 : 0;
        int segState = isActive ? !comState : comState;
        if (segState) bshrC |= (1 << (4 + s));
        else bcrC |= (1 << (4 + s));
      }
      
      GPIOC->BSHR = bshrC;
      GPIOC->BCR = bcrC;
      
      // 液晶をしっかり点灯させるため純粋なディレイに復旧(電圧降下防止)
      delayMicroseconds(500);
    }
  }
  
  // 終了後は全LCDピンを完全フローティングに戻し、DC印加を防ぐ
  GPIOA->CFGLR = (GPIOA->CFGLR & ~(0xFu << (2 * 4))) | (CH_IN_FL << (2 * 4));
  GPIOC->CFGLR = (GPIOC->CFGLR & ~((0xFu << (1 * 4)) | (0xFu << (2 * 4)) | (0xFu << (4 * 4)) | (0xFu << (5 * 4)) | (0xFu << (6 * 4)))) 
               | ((CH_IN_FL << (1 * 4)) | (CH_IN_FL << (2 * 4)) | (CH_IN_FL << (4 * 4)) | (CH_IN_FL << (5 * 4)) | (CH_IN_FL << (6 * 4)));
}

void WaitAndDriveLCD(int ms) {
  unsigned long start = millis();
  while (millis() - start < (unsigned long)ms) {
    UpdateLCD(dispNum);
  }
}

void blinkLed(int times) {
  for (int i = 0; i < times; i++) {
    GPIOC->BSHR = 1;
    WaitAndDriveLCD(120);
    GPIOC->BCR = 1;
    WaitAndDriveLCD(120);
  }
}

int adcSampleTouch(int bit, int isFirst) {
  uint32_t sh = bit * 4;
  
  // 全ピンを連動放電(巨大アンテナ化)
  GPIOA->CFGLR &= ~(0xFu << sh);
  GPIOA->CFGLR |=  (CH_OUT << sh);
  GPIOA->BCR = (1 << bit);
  
  GPIOC->CFGLR &= ~((0xFu << 4) | (0xFu << 8) | (0xFu << 16) | (0xFu << 20) | (0xFu << 24));
  GPIOC->CFGLR |=  ((CH_OUT << 4) | (CH_OUT << 8) | (CH_OUT << 16) | (CH_OUT << 20) | (CH_OUT << 24));
  GPIOC->BCR = (1 << 1) | (1 << 2) | (1 << 4) | (1 << 5) | (1 << 6);
  
  // 液晶の電荷リセットは32回サンプリングの「初回のみ」長く取り、暗転時間を最小化する
  if (isFirst) {
    delayMicroseconds(1000);
  } else {
    delayMicroseconds(80);
  }
  
  // 全ピンを入出力プルアップ(アンテナ化)へ切替
  GPIOA->CFGLR &= ~(0xFu << sh);
  GPIOA->CFGLR |=  (CH_IN_PU << sh);
  GPIOA->OUTDR |= (1 << bit);
  
  GPIOC->CFGLR &= ~((0xFu << 4) | (0xFu << 8) | (0xFu << 16) | (0xFu << 20) | (0xFu << 24));
  GPIOC->CFGLR |=  ((CH_IN_PU << 4) | (CH_IN_PU << 8) | (CH_IN_PU << 16) | (CH_IN_PU << 20) | (CH_IN_PU << 24));
  GPIOC->OUTDR |= (1 << 1) | (1 << 2) | (1 << 4) | (1 << 5) | (1 << 6);
  
  // SENSE_BIT(PA2)のADCサンプリング
  ADC1->CTLR2 = ADC_SWSTART | ADC_ADON | ADC_EXTSEL;
  while (!(ADC1->STATR & ADC_EOC));
  return ADC1->RDATAR;
}

int readCap(int bit) {
  long s = 0;
  for (int i = 0; i < 32; i++) s += adcSampleTouch(bit, (i == 0) ? 1 : 0);
  return (int)s;
}

int isTouch(int raw) {
  int d = baseA0 - raw;
  if (d < 0) d = 0;
  return (d >= TOUCH_MIN) ? 1 : 0;
}

int rawToPercent(int raw) {
  int d_raw = baseA0 - raw;
  int d_max = baseA0 - maxVal;
  
  if (d_max <= 0) return 100;
  
  long p = ((long)d_raw * 100L) / (long)d_max;
  if (p < 0) p = 0;
  if (p > 100) p = 100;
  return (int)p;
}

void smoothReset() {
  smoothIdx = 0;
  smoothCount = 0;
  smoothSum = 0;
}

int smoothPush(int raw) {
  if (smoothCount < SMOOTH_N) {
    smoothBuf[smoothIdx] = raw;
    smoothSum += raw;
    smoothCount++;
  } else {
    smoothSum -= smoothBuf[smoothIdx];
    smoothSum += raw;
    smoothBuf[smoothIdx] = raw;
  }
  smoothIdx = (smoothIdx + 1) % SMOOTH_N;
  return (int)(smoothSum / smoothCount);
}

void setup() {
  setupHardwareUart();
  RCC->APB2PCENR |= (RCC_APB2Periph_GPIOA | RCC_APB2Periph_GPIOC);
  GPIOC->CFGLR &= ~0xFu;
  GPIOC->CFGLR |= CH_OUT;
  setupPwmPc3();
  setPwmPercent(0);
  setupAdcTouch();
  
  uartSendString("=== LCD slider Ver0.26 (ADC touch & LCD test) ===\r\n");
  uartSendString("Calibrating OFF...\r\n");
  WaitAndDriveLCD(400);
  
  for (int i = 0; i < 5; i++) {
    readCap(SENSE_BIT);
    WaitAndDriveLCD(20);
  }
  
  baseA0 = readCap(SENSE_BIT);
  uartSendString("Base A0=");
  uartSendInt(baseA0);
  uartSendString("\r\n");
  
  blinkLed(1);
  calState = 1;
  uartSendString("Touch MAX\r\n");
}

void loop() {
  int raw = readCap(SENSE_BIT);
  unsigned long now = millis();
  
  if (calState == 1) {
    if (isTouch(raw)) {
      phaseStart = now;
      calSum = 0;
      calCount = 0;
      calState = 2;
    }
    WaitAndDriveLCD(20);
    return;
  }
  
  if (calState == 2) {
    if (!isTouch(raw)) {
      calState = 1;
      WaitAndDriveLCD(20);
      return;
    }
    unsigned long elapsed = now - phaseStart;
    if (elapsed >= WIN_START_MS && elapsed <= STABLE_MS) {
      calSum += raw;
      calCount++;
    }
    if (elapsed < STABLE_MS) {
      WaitAndDriveLCD(20);
      return;
    }
    maxVal = (calCount > 0) ? (int)(calSum / calCount) : raw;
    uartSendString("MAX=");
    uartSendInt(maxVal);
    uartSendString("\r\n");
    blinkLed(2);
    phaseStart = millis();
    calState = 3;
    uartSendString("Release wait\r\n");
    WaitAndDriveLCD(20);
    return;
  }
  
  if (calState == 3) {
    if (isTouch(raw)) {
      phaseStart = now;
    } else if ((now - phaseStart) >= STABLE_MS) {
      calState = 4;
      smoothReset();
      uartSendString("RUN\r\n");
    }
    WaitAndDriveLCD(20);
    return;
  }
  
  int currentState = 0;
  const char* pos = "NONE ";
  int sliderPos = 0;
  
  if (!isTouch(raw)) {
    currentState = 0;
    pos = "NONE ";
    smoothReset();
    GPIOC->BCR = 1;
    setPwmPercent(0);
  } else {
    currentState = 1;
    pos = "TOUCH";
    int avgRaw = smoothPush(raw);
    sliderPos = rawToPercent(avgRaw);
    GPIOC->BSHR = 1;
    setPwmPercent(sliderPos);
  }
  
  // タッチした瞬間(エッジ)を検出して表示数字をインクリメント
  static int lastCurrentState = 0;
  if (currentState == 1 && lastCurrentState == 0) {
    dispNum = (dispNum + 1) % 16;
  }
  lastCurrentState = currentState;
  
  static int printTimer = 0;
  printTimer++;
  if (printTimer >= 50) {
    uartSendString("raw=");
    uartSendInt(raw);
    uartSendString(" d=");
    uartSendInt(baseA0 - raw);
    uartSendChar(' ');
    uartSendString(pos);
    
    if (currentState != 0) {
      uartSendString(" POS=");
      uartSendInt(sliderPos);
      uartSendString("%");
    }
    
    uartSendString("\r\n");
    printTimer = 0;
  }
  
  WaitAndDriveLCD(20);
}
